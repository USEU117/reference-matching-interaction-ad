# --- 0. 先释放内存 -----------------------------------------------------------
# 关闭浏览器/桌面应用，使可用物理内存 >= 6 GiB（15.8 GiB 机器）
# 查看: Get-CimInstance Win32_OperatingSystem | Select-Object FreePhysicalMemory

# --- 1. 前置检查（只读，不会启动运行） ---------------------------------------
powershell -ExecutionPolicy Bypass -File experiments\prereg_20260924\state\A08_preflight.ps1
# 可用内存紧张时放宽阈值:  ... A08_preflight.ps1 -MinFreeMiB 3000
# 必须看到 "=== PRE-FLIGHT OK ===" 才继续。

# --- 2. 主命令（单进程，可随时被杀，最多丢当前一个类别） ---------------------
.venv-anomalyclip\Scripts\python.exe -u ^
  scripts\limitation_closure_20260915\e1_fullpixel_ci.py ^
  --mode run --resume --stride 1 --replicates 1000 ^
  --datasets mpdd btad ^
  --output experiments\prereg_20260924\out\A08
#   --resume : 扫描 out\A08\units\，跳过已完成的单元/类别，只跑剩余
#   重复执行同一条命令即可断点续跑；全部完成后再执行一次只会做汇总。
#   建议设置 E1_CHUNK 保持默认 4096（改 chunk 会使已有断点失效并重算）。

# --- 3. 只看进度 -------------------------------------------------------------
Get-Content experiments\prereg_20260924\logs\A08_stride1.out -Tail 5 -Wait

#   experiments\prereg_20260924\out\A08\replicate_stride1.npz
#   experiments\prereg_20260924\out\A08\point_stride1.csv
#   experiments\prereg_20260924\out\A08\E1_STATUS_stride1.json
#   experiments\prereg_20260924\out\A08\units\<dataset>_s<seed>_k<shot>.json  (断点)
